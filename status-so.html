// ========================================== // RENDER DYNAMIC TABLE DENGAN FREEZE & SORT // ========================================== function renderSOTable() { const savedString = localStorage.getItem('savedDashboardData'); const tbody = document.getElementById('so-table-body'); const thead = document.getElementById('so-table-head'); if (!savedString || !tbody || !thead) return; try { let rawData = JSON.parse(savedString); if (!rawData || rawData.length === 0) return; // Tambahkan field original index untuk mengembalikan urutan "No" jika reset rawData = rawData.map((row, idx) => ({ ...row, _originalIndex: idx + 1 })); const allowedColumns = [ "SO INDUK", "SO PENAMBAH", "NAMA CLIENT", "NAMA AREA", "NAMA JOB", "TGL APPROVE", "ASSIGN RO", "AGING-SS", "SLA LT OFFERING", "GROUP SLA BELUM OFFERING", "DUE DATE FULFILLMENT", "AGING NON PENDING", "TANGGAL INVALID - PENDING", "TASK ID", "RO LINE", "AGING UNFULFILL", "PARTICIPATE", "KATEGORI FULFILLMENT" ]; // SORTING LOGIC SEBELUM RENDER if (window.currentSort.key) { rawData.sort((a, b) => { if (window.currentSort.key === 'No') { return window.currentSort.asc ? a._originalIndex - b._originalIndex : b._originalIndex - a._originalIndex; } const keyA = Object.keys(a).find(k => k.trim().toUpperCase() === window.currentSort.key.toUpperCase()); const keyB = Object.keys(b).find(k => k.trim().toUpperCase() === window.currentSort.key.toUpperCase()); let valA = keyA ? a[keyA] : ""; let valB = keyB ? b[keyB] : ""; if (window.currentSort.key === "DUE DATE FULFILLMENT" || window.currentSort.key === "TGL APPROVE") { valA = parseDateForSort(valA); valB = parseDateForSort(valB); } else if (!isNaN(valA) && !isNaN(valB) && valA !== "" && valB !== "") { valA = Number(valA); valB = Number(valB); } else { valA = String(valA).toLowerCase(); valB = String(valB).toLowerCase(); } if (valA < valB) return window.currentSort.asc ? -1 : 1; if (valA > valB) return window.currentSort.asc ? 1 : -1; return 0; }); } // BUAT HEADER THEAD let headHTML = ''; const noIcon = window.currentSort.key === 'No' ? (window.currentSort.asc ? "fa-sort-up text-teal-600" : "fa-sort-down text-teal-600") : "fa-sort opacity-30"; headHTML += `No `; allowedColumns.forEach((key, i) => { let stickyClasses = "cursor-pointer hover:bg-slate-200 transition-colors sticky top-0 z-20 bg-slate-50"; if (i === 0) stickyClasses += " left-[60px] bg-slate-100 z-30 sticky-col"; let sortIcon = window.currentSort.key === key ? (window.currentSort.asc ? "fa-sort-up text-teal-600" : "fa-sort-down text-teal-600") : "fa-sort opacity-30"; headHTML += `${key} `; }); headHTML += ''; thead.innerHTML = headHTML; const today = new Date(); today.setHours(0,0,0,0); const assignRoSet = new Set(); const kategoriSet = new Set(); const slaSet = new Set(); let bodyHTML = ''; rawData.forEach((row) => { const rowKeys = Object.keys(row); const assignRoKey = rowKeys.find(k => k.trim().toUpperCase() === "ASSIGN RO"); const kategoriKey = rowKeys.find(k => k.trim().toUpperCase() === "KATEGORI FULFILLMENT"); const slaKey = rowKeys.find(k => k.trim().toUpperCase() === "SLA LT OFFERING"); const dueDateKey = rowKeys.find(k => k.trim().toUpperCase() === "DUE DATE FULFILLMENT"); // --- [BARU] EKSTRAK DATA UNTUK KALKULASI PERSENTASE --- const finishKey = rowKeys.find(k => k.trim().toLowerCase() === 'finish'); const unfulfillKey = rowKeys.find(k => k.trim().toLowerCase() === 'unfulfill'); const offeringKey = rowKeys.find(k => k.trim().toLowerCase() === 'offering'); const finishVal = finishKey ? (parseInt(row[finishKey]) || 0) : 0; const unfulfillVal = unfulfillKey ? (parseInt(row[unfulfillKey]) || 0) : 0; const offeringVal = offeringKey ? (parseInt(row[offeringKey]) || 0) : 0; const roValue = (assignRoKey && row[assignRoKey]) ? String(row[assignRoKey]).trim() : ''; const kategoriValue = (kategoriKey && row[kategoriKey]) ? String(row[kategoriKey]).trim() : ''; const slaValue = (slaKey && row[slaKey]) ? String(row[slaKey]).trim() : ''; const dueDateStr = (dueDateKey && row[dueDateKey]) ? String(row[dueDateKey]).trim() : ''; if (roValue) assignRoSet.add(roValue); if (kategoriValue) kategoriSet.add(kategoriValue); if (slaValue) slaSet.add(slaValue); let rowClass = "hover:bg-slate-50/80 transition-colors border-b border-slate-100 group"; let dueBadge = ''; let statusBadgeAttr = ''; if (dueDateStr && dueDateStr !== '-') { let dueDateObj = null; if (dueDateStr.includes('/')) { const parts = dueDateStr.split('/'); if (parts.length === 3) dueDateObj = new Date(parts[2], parts[1] - 1, parts[0]); } else { dueDateObj = new Date(dueDateStr); } if (dueDateObj && !isNaN(dueDateObj)) { const diffDays = Math.ceil((dueDateObj - today) / (1000 * 60 * 60 * 24)); if (diffDays < 0) { rowClass = "bg-rose-50 hover:bg-rose-100 transition-colors border-b border-rose-200 group text-slate-900"; dueBadge = ` Overdue`; statusBadgeAttr = 'overdue'; } else if (diffDays <= 3) { rowClass = "bg-orange-50 hover:bg-orange-100 transition-colors border-b border-orange-200 group text-slate-900"; dueBadge = ` H-${diffDays}`; statusBadgeAttr = 'h3'; } } } // --- [BARU] SUSUPKAN METRIK KE DALAM ATTRIBUTE TR --- bodyHTML += ``; bodyHTML += `${row._originalIndex}`; allowedColumns.forEach((key, i) => { const matchedKey = rowKeys.find(k => k.trim().toUpperCase() === key.toUpperCase()); const cellData = (matchedKey && row[matchedKey]) ? row[matchedKey] : '-'; let cellClasses = "p-4 text-slate-700 whitespace-nowrap group-hover:text-slate-900 font-medium"; if (i === 0) cellClasses += " sticky left-[60px] bg-white group-hover:bg-slate-50 z-10 sticky-col"; if (key === "DUE DATE FULFILLMENT") { bodyHTML += `${cellData} ${dueBadge}`; } else { bodyHTML += `${cellData}`; } }); bodyHTML += ''; }); tbody.innerHTML = bodyHTML; const fillDropdown = (id, set) => { const el = document.getElementById(id); if (el && el.options.length <= 1) { const current = el.value; Array.from(set).sort().forEach(val => el.innerHTML += ``); el.value = current; } } fillDropdown('filter-assign-ro', assignRoSet); fillDropdown('filter-kategori', kategoriSet); fillDropdown('filter-sla', slaSet); applyFilters(); } catch (error) { console.error("Gagal merender tabel", error); } } window.renderSOTable = renderSOTable; // ========================================== // LOGIKA FILTER & UPDATE MINI INSIGHTS + KARTU PERSENTASE // ========================================== function applyFilters() { const searchInput = document.getElementById('search-csv'); const filterInput = document.getElementById('filter-assign-ro'); const filterKategoriInput = document.getElementById('filter-kategori'); const filterSlaInput = document.getElementById('filter-sla'); const tbody = document.getElementById('so-table-body'); if (!tbody) return; const searchTerm = searchInput ? searchInput.value.toLowerCase() : ""; const filterRo = filterInput ? filterInput.value : ""; const filterKat = filterKategoriInput ? filterKategoriInput.value : ""; const filterSla = filterSlaInput ? filterSlaInput.value : ""; const rows = tbody.getElementsByTagName('tr'); let visibleCount = 0; let overdueCount = 0; let h3Count = 0; // --- [BARU] Variabel untuk menampung metrik dinamis --- let terFulfill = 0, belumFulfill = 0, sudahOffering = 0, belumOffering = 0; Array.from(rows).forEach(row => { if (row.children.length === 1) return; // Skip baris "menunggu data" const text = row.innerText.toLowerCase(); const rowRo = row.getAttribute('data-assign-ro') || ""; const rowKat = row.getAttribute('data-kategori') || ""; const rowSla = row.getAttribute('data-sla') || ""; const statusBadge = row.getAttribute('data-status-badge') || ""; const matchSearch = text.includes(searchTerm); const matchRo = filterRo === "" || rowRo === filterRo; const matchKat = filterKat === "" || rowKat === filterKat; const matchSla = filterSla === "" || rowSla === filterSla; if (matchSearch && matchRo && matchKat && matchSla) { row.style.display = ''; visibleCount++; if (statusBadge === 'overdue') overdueCount++; if (statusBadge === 'h3') h3Count++; // --- [BARU] Kalkulasi angka dari baris yang berhasil ter-filter --- const finish = parseInt(row.getAttribute('data-finish')) || 0; const unfulfill = parseInt(row.getAttribute('data-unfulfill')) || 0; const offering = parseInt(row.getAttribute('data-offering')) || 0; terFulfill += finish; belumFulfill += unfulfill; if (finish === 1 || unfulfill === 1) { if (offering > 0) sudahOffering++; else belumOffering++; } } else { row.style.display = 'none'; } }); // Perbarui Mini Insights (Baris, Overdue, H-3) document.getElementById('total-row-count').innerText = `${visibleCount} Baris`; document.getElementById('insight-overdue').innerHTML = `${overdueCount} Overdue`; document.getElementById('insight-h3').innerHTML = `${h3Count} H-3`; // --- [BARU] Perbarui 4 Kartu Metrik Utama --- const totalFulfill = terFulfill + belumFulfill; const totalOffering = sudahOffering + belumOffering; document.getElementById('text-ter-fulfill').innerText = `${terFulfill} (${totalFulfill > 0 ? Math.round((terFulfill / totalFulfill) * 100) : 0}%)`; document.getElementById('text-belum-fulfill').innerText = `${belumFulfill} (${totalFulfill > 0 ? Math.round((belumFulfill / totalFulfill) * 100) : 0}%)`; document.getElementById('text-sudah-offering').innerText = `${sudahOffering} (${totalOffering > 0 ? Math.round((sudahOffering / totalOffering) * 100) : 0}%)`; document.getElementById('text-belum-offering').innerText = `${belumOffering} (${totalOffering > 0 ? Math.round((belumOffering / totalOffering) * 100) : 0}%)`; } window.applyFilters = applyFilters;
